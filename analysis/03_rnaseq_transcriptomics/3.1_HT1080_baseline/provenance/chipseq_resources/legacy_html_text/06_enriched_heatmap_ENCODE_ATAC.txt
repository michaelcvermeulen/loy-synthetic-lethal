get_gene_sets <- function(de, fdr) {
  list(
    up   = de$gene_name[de$FDR < fdr & de$logFC > 0],
    down = de$gene_name[de$FDR < fdr & de$logFC < 0],
    nde  = de$gene_name[de$FDR >= fdr]
  )
}

for (fdr in fdr_thresholds) {

  summ <- compute_summary(fdr)

  # Matrix for heatmap: rows = tissues, columns = UP/DOWN/Non-DE
  hm_mat <- as.matrix(summ[, c("mean_up", "mean_down", "mean_nde")])
  rownames(hm_mat) <- summ$tissue
  colnames(hm_mat) <- c("Up in LOY", "Down in LOY", "Non-DE")

  # Row annotation: classification
  cls_vec <- summ$classification
  cls_vec[is.na(cls_vec)] <- "unknown"
  row_anno <- rowAnnotation(
    Classification = cls_vec,
    col = list(Classification = classification_colors),
    show_legend = TRUE
  )

  col_fun <- colorRamp2(
    quantile(hm_mat, c(0, 0.5, 1), na.rm = TRUE),
    c("white", "#FF7F00", "#8B0000")
  )

  ht <- Heatmap(hm_mat,
    name = "Mean ATAC\n(fold change)",
    col = col_fun,
    cluster_columns = FALSE,
    clustering_distance_rows = "euclidean",
    clustering_method_rows = "ward.D2",
    row_names_gp = gpar(fontsize = 6),
    column_names_gp = gpar(fontsize = 10),
    left_annotation = row_anno,
    column_title = sprintf("Mean ATAC-seq Signal at TSS (±5kb) — FDR < %s", fdr),
    row_title = "Tissue / Cell Type",
    heatmap_legend_param = list(title_gp = gpar(fontsize = 8))
  )

  draw(ht)
  cat("\n\n")

  pdf(file.path(figures_dir, sprintf("ENCODE_ATAC_summary_heatmap_fdr%s.pdf", fdr)),
      width = 10, height = 20)
  draw(ht)
  dev.off()
}

Permutation Analysis — Top 5 Tissues
for (fdr in fdr_thresholds) {
  cat(sprintf("\n\n## FDR < %s\n\n", fdr))

  summ <- compute_summary(fdr) %>% filter(!is.na(diff_up_nde))
  top5 <- summ %>% arrange(desc(diff_up_nde)) %>% head(5) %>% pull(tissue)

  for (tissue in top5) {
    cat(sprintf("\n\n### %s\n\n", tissue))

    result <- run_permutation(tissue, fdr)
    if (is.null(result)) {
      cat("Too few DE genes for permutation test.\n\n")
      next
    }

    df_perm <- result$plot_df %>% filter(type == "permutation")
    df_agg  <- result$plot_df %>% filter(type == "aggregate")
    df_obs  <- result$plot_df %>% filter(type == "observed")

    p <- ggplot() +
      geom_line(data = df_perm,
                aes(x = position, y = signal, group = interaction(direction, iter)),
                color = "grey60", alpha = 0.05, linewidth = 0.3) +
      geom_line(data = df_agg,
                aes(x = position, y = signal),
                color = "black", linewidth = 0.8) +
      geom_line(data = df_obs,
                aes(x = position, y = signal, color = direction),
                linewidth = 1.2) +
      geom_vline(xintercept = 0, linetype = "dashed", color = "grey40") +
      facet_wrap(~direction, scales = "free_y", ncol = 2) +
      scale_color_manual(values = obs_colors) +
      theme_minimal(base_size = 12) +
      labs(title = sprintf("Permutation Test: %s (n = %d permutations)", tissue, n_perm),
           subtitle = sprintf(
             "FDR < %s — Grey = random (N_up=%d, N_down=%d); Black = aggregate; Colored = observed",
             fdr, result$n_up, result$n_down),
           x = "Distance from TSS (bp)",
           y = "Mean ATAC signal (fold change)",
           color = "") +
      theme(legend.position = "bottom",
            strip.text = element_text(face = "bold"))
    print(p)
    cat("\n\n")
  }

  # Save combined PDF for top 5
  ggsave(file.path(figures_dir, sprintf("ENCODE_ATAC_permutation_top5_fdr%s.pdf", fdr)),
         plot = p, width = 14, height = 6)
}
