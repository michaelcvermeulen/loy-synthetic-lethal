Contrast direction
DiffBind alphabetizes condition labels, so its native
Fold
is
log2(CTRL/LOY)
. We flip the sign in
this pipeline so that
positive
Fold
= gained in
LOY
.

Volcano without chrY
chrY peaks are systematically lost in LOY by definition (it’s the
chromosome that is
missing
), and they dominate the LOY-lost
side of the canonical volcano in raw numbers despite being only ~0.1% of
the consensus set. Removing them shows the “real” LOY effect across the
rest of the genome — what would be visible if LOY were a phenotype
rather than a chromosomal event.
This is a
filter on the rendered DAR table only
—
DESeq2 is not re-run, FDR is not re-adjusted. q-values are identical to
the canonical view; only the displayed point set changes.
volc_df <- dar_full[dar_full$seqnames != "chrY", ]
volc_df$signif <- with(volc_df,
                       !is.na(FDR) & FDR < fdr_thresh & abs(Fold) > lfc_thresh)
volc_df$direction <- with(volc_df,
                          ifelse(!signif, "ns",
                          ifelse(Fold > 0, "Gained_in_LOY", "Lost_in_LOY")))
top_label <- volc_df[order(volc_df$FDR, -abs(volc_df$Fold)), ][seq_len(min(25, nrow(volc_df))), ]
top_label$label <- paste0(top_label$seqnames, ":",
                          format(top_label$start, big.mark = ","))

p_volc_noY <- ggplot2::ggplot(volc_df,
    ggplot2::aes(x = Fold, y = -log10(pmax(FDR, 1e-300)),
                 color = direction)) +
  ggplot2::geom_point(alpha = 0.5, size = 0.6) +
  ggplot2::scale_color_manual(values = c(Gained_in_LOY = "#D55E00",
                                         Lost_in_LOY   = "#0072B2",
                                         ns            = "grey80")) +
  ggplot2::geom_vline(xintercept = c(-lfc_thresh, lfc_thresh),
                      linetype = "dashed", color = "grey40") +
  ggplot2::geom_hline(yintercept = -log10(fdr_thresh),
                      linetype = "dashed", color = "grey40") +
  ggrepel::geom_text_repel(data = top_label,
                           ggplot2::aes(label = label),
                           size = 2.5, max.overlaps = 30,
                           color = "black") +
  ggplot2::labs(x = "log2 Fold Change (LOY vs CTRL)",
                y = "-log10 FDR",
                title = "HT1080 ATAC: LOY vs CTRL DARs (chrY excluded)",
                subtitle = sprintf("chrY peaks excluded: %d removed, %d shown",
                                   sum(dar_full$seqnames == "chrY"),
                                   nrow(volc_df)),
                color = NULL) +
  ggplot2::theme_bw(base_size = 11)
ggsave_both("volcano_plot_no_chrY", p_volc_noY, width = 7.5, height = 5.5)
