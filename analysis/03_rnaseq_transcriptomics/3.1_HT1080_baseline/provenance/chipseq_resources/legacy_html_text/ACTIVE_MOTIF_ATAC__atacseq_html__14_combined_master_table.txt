  atac_best_FDR = signif(atac_best_FDR, 3),
  atac_best = atac_best_annotation,
  n_open = atac_n_open, n_closed = atac_n_closed,
  atac_open_log2FC = round(atac_open_log2FC, 3),
  atac_open_FDR = signif(atac_open_FDR, 3),
