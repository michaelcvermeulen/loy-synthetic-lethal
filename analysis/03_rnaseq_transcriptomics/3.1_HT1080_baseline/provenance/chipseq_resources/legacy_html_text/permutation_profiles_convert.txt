Permutation Analysis
n_perm <- 500
position_vec <- seq(-extend_bp, extend_bp - 1,
                    length.out = ncol(mat_list[[1]]))

                n_perm, n_up, n_down))

# Pre-allocate list for permutation results
perm_list <- list()

for (mark in names(mat_list)) {
  mat <- as.matrix(mat_list[[mark]])

  # Observed profiles
  obs_up   <- colMeans(mat[up_idx, , drop = FALSE], na.rm = TRUE)
  obs_down <- colMeans(mat[down_idx, , drop = FALSE], na.rm = TRUE)

  # Aggregate (all genes)
  agg <- colMeans(mat, na.rm = TRUE)

  # Permutation profiles
  perm_up_mat   <- matrix(NA_real_, nrow = n_perm, ncol = ncol(mat))
  perm_down_mat <- matrix(NA_real_, nrow = n_perm, ncol = ncol(mat))

  for (i in seq_len(n_perm)) {
    idx_up   <- sample(n_total, n_up)
    idx_down <- sample(n_total, n_down)
    perm_up_mat[i, ]   <- colMeans(mat[idx_up, , drop = FALSE], na.rm = TRUE)
    perm_down_mat[i, ] <- colMeans(mat[idx_down, , drop = FALSE], na.rm = TRUE)
  }

  # Build data frames for this mark
  # Permutation cloud
  for (i in seq_len(n_perm)) {
    perm_list[[length(perm_list) + 1]] <- data.frame(
      mark = mark, direction = "Up in LOY", type = "permutation",
      iter = i, position = position_vec, signal = perm_up_mat[i, ],
      stringsAsFactors = FALSE
    )
    perm_list[[length(perm_list) + 1]] <- data.frame(
      mark = mark, direction = "Down in LOY", type = "permutation",
      iter = i, position = position_vec, signal = perm_down_mat[i, ],
      stringsAsFactors = FALSE
    )
  }

  # Observed
  perm_list[[length(perm_list) + 1]] <- data.frame(
    mark = mark, direction = "Up in LOY", type = "observed",
    iter = 0, position = position_vec, signal = obs_up,
    stringsAsFactors = FALSE
  )
  perm_list[[length(perm_list) + 1]] <- data.frame(
    mark = mark, direction = "Down in LOY", type = "observed",
    iter = 0, position = position_vec, signal = obs_down,
    stringsAsFactors = FALSE
  )

  # Aggregate
  perm_list[[length(perm_list) + 1]] <- data.frame(
    mark = mark, direction = "Up in LOY", type = "aggregate",
    iter = 0, position = position_vec, signal = agg,
    stringsAsFactors = FALSE
  )
  perm_list[[length(perm_list) + 1]] <- data.frame(
    mark = mark, direction = "Down in LOY", type = "aggregate",
    iter = 0, position = position_vec, signal = agg,
    stringsAsFactors = FALSE
  )
}

plot_df <- rbindlist(perm_list) %>% as_tibble()
