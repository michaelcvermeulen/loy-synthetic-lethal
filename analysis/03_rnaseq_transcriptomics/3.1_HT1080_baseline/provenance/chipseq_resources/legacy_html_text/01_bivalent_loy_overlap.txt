Fisher’s Exact Enrichment
# Universe: genes present in both DE and promoter annotation

# Run Fisher's exact test for each chromatin state
states <- list(
  Bivalent = bivalent_genes,
  `H3K4me3-only` = k4_only_genes,
  `H3K27me3-only` = k27_only_genes
)

fisher_results <- lapply(names(states), function(state_name) {
  state_genes <- intersect(states[[state_name]], universe)
  non_state <- setdiff(universe, state_genes)

  a <- sum(state_genes %in% sig_genes)     # state + DE
  b <- length(state_genes) - a              # state + not DE
  c <- sum(non_state %in% sig_genes)        # not state + DE
  d <- length(non_state) - c                # not state + not DE

  mat <- matrix(c(a, b, c, d), nrow = 2,
                dimnames = list(c("DE", "Not DE"), c("State", "Other")))
  ft <- fisher.test(mat)

  data.frame(
    chromatin_state = state_name,
    n_state = length(state_genes),
    n_de_in_state = a,
    pct_de_in_state = round(a / length(state_genes) * 100, 1),
    n_de_outside = c,
    pct_de_outside = round(c / length(non_state) * 100, 1),
    odds_ratio = ft$estimate,
    or_lower = ft$conf.int[1],
    or_upper = ft$conf.int[2],
    p_value = ft$p.value,
    stringsAsFactors = FALSE
  )
})

fisher_df <- bind_rows(fisher_results)
fisher_df$padj <- p.adjust(fisher_df$p_value, method = "BH")
